IT Equipment Manager

เชื่อมต่อ Google Apps Script เพื่อเริ่มใช้งาน

1

เปิด Google Apps Script

ไปที่ script.google.com → คลิก New Project

2

วางโค้ดลงใน Code.gs

ลบโค้ดเดิมออก แล้ว copy โค้ดด้านล่างนี้ทั้งหมดวางแทน

/** * IT Equipment Manager — Google Apps Script Backend * Deploy: Execute as "Me" | Who has access: Anyone */ const FOLDER_NAME = 'IT_Equipment_Manager'; function doGet(e) { try { const action = (e.parameter && e.parameter.action) || 'test'; if (action === 'test') return respond({status:'ok',message:'API พร้อมใช้งาน ✓',ts:new Date().toISOString()}); if (action === 'getAll') return respond({docs:getAllDocs()}); if (action === 'getDoc') return respond({doc:getDocById(e.parameter.id)}); return respond({error:'Unknown action'}); } catch(ex) { return respond({error:ex.toString()}); } } function doPost(e) { try { const b = JSON.parse(e.postData.contents); if (b.action === 'saveDoc') return respond(saveDoc(b.doc)); if (b.action === 'deleteDoc') return respond(deleteDoc(b.id)); return respond({error:'Unknown action'}); } catch(ex) { return respond({error:ex.toString()}); } } function respond(data) { const o = ContentService.createTextOutput(JSON.stringify(data)); o.setMimeType(ContentService.MimeType.JSON); return o; } function getFolder() { const i = DriveApp.getFoldersByName(FOLDER_NAME); return i.hasNext() ? i.next() : DriveApp.createFolder(FOLDER_NAME); } function getAllDocs() { const folder = getFolder(), files = folder.getFiles(), docs = []; while (files.hasNext()) { const f = files.next(); if (f.getName().endsWith('.json')) { try { docs.push(JSON.parse(f.getBlob().getDataAsString('utf-8'))); } catch(e){} } } return docs.sort((a,b)=>new Date(b.createdAt||0)-new Date(a.createdAt||0)); } function getDocById(id) { if (!id) return null; const f = getFolder().getFilesByName(id+'.json'); return f.hasNext() ? JSON.parse(f.next().getBlob().getDataAsString('utf-8')) : null; } function saveDoc(doc) { if (!doc||!doc.id) throw new Error('Document must have id'); const lock = LockService.getScriptLock(); if (!lock.tryLock(20000)) throw new Error('Server busy, retry'); try { const folder = getFolder(), fn = doc.id+'.json', c = JSON.stringify(doc); const files = folder.getFilesByName(fn); files.hasNext() ? files.next().setContent(c) : folder.createFile(fn, c, MimeType.PLAIN_TEXT); return {saved:true, id:doc.id}; } finally { lock.releaseLock(); } } function deleteDoc(id) { if (!id) throw new Error('id required'); const f = getFolder().getFilesByName(id+'.json'); if (!f.hasNext()) throw new Error('Not found: '+id); f.next().setTrashed(true); return {deleted:true, id}; }
3

Deploy เป็น Web App

คลิก Deploy → New Deployment → Web App
ตั้งค่า: Execute as: Me | Who has access: Anyone
กด Deploy → Allow → Copy Web App URL

4

วาง Web App URL ด้านล่าง

URL จะมีรูปแบบ: https://script.google.com/macros/s/.../exec
